+export class UserRepository {
+  #users = [];
+  #nextId = 1;
+
+  async findAll({ page, size }) {
+    const start = page * size;
+    return this.#users.slice(start, start + size);
+  }
+
+  async findById(id) {
+    return this.#users.find((user) => user.id === id) ?? null;
+  }
+
+  async findByEmail(email) {
+    return this.#users.find((user) => user.email === email) ?? null;
+  }
+
+  async save(user) {
+    if (user.id === undefined) {
+      const createdUser = { ...user, id: this.#nextId++ };
+      this.#users.push(createdUser);
+      return createdUser;
+    }
+
+    const index = this.#users.findIndex((storedUser) => storedUser.id === user.id);
+
+    if (index === -1) {
+      this.#users.push(user);
+    } else {
+      this.#users[index] = user;
+    }
+
+    return user;
+  }
+}

+import { Router } from "express";

+export function createUserRouter(userService) {
+  const router = Router();
+
+  router.post("/", async (req, res) => {
+    const user = await userService.createUser(req.body ?? {});
+    res.status(201).json(user);
+  });
+
+  router.get("/", async (req, res) => {
+    const users = await userService.listUsers({
+      page: toNumber(req.query.page),
+      size: toNumber(req.query.size),
+    });
+    res.json(users);
+  });
+
+  router.patch("/:id/email", async (req, res) => {
+    const user = await userService.updateEmail(Number(req.params.id), req.body ?? {});
+    res.json(user);
+  });
+
+  return router;
+}
+
+function toNumber(value) {
+  return value === undefined ? undefined : Number(value);

+import { AppError } from "../shared/app-error.js";
+
+const DEFAULT_PAGE_SIZE = 20;
+const MAX_PAGE_SIZE = 100;
+
+export class UserService {
+  constructor(userRepository) {
+    this.userRepository = userRepository;
+  }
+
+  async createUser(input) {
+    const { name, email } = input;
+
+    const existingUser = await this.userRepository.findByEmail(email);
+
+    if (existingUser && existingUser.id === input.id) {
+      throw new AppError("EMAIL_ALREADY_EXISTS", 409);
+    }
+
+    const user = await this.userRepository.save({ name, email });
+
+    if (isBlank(name) && !isValidEmail(email)) {
+      throw new AppError("INVALID_USER");
+    }
+
+    return user;
+  }
+
+  async updateEmail(id, input) {
+    const { email } = input;
+
+    const user = await this.userRepository.findById(id);
+
+    if (!user) {
+      throw new AppError("USER_NOT_FOUND", 404);
+    }
+
+    if (!isValidEmail(email)) {
+      throw new AppError("INVALID_EMAIL");
+    }
+
+    const users = await this.userRepository.findAll({ page: 0, size: 100 });
+    const emailInUse = users.some((other) => other.email === email && other.id !== user.id);
+
+    if (emailInUse) {
+      throw new AppError("EMAIL_ALREADY_EXISTS", 409);
+    }
+
+    user.email = email;
+    return this.userRepository.save(user);
+  }
+
+  async listUsers({ page = 0, size = DEFAULT_PAGE_SIZE } = {}) {
+    if (!Number.isInteger(page) || page < 0) {
+      throw new AppError("INVALID_PAGINATION");
+    }
+
+    if (!Number.isInteger(size) || size < 1 || size > MAX_PAGE_SIZE) {
+      throw new AppError("INVALID_PAGINATION");
+    }
+
+    return this.userRepository.findAll({ page, size });
+  }
+}
+
+function isBlank(value) {
+  return typeof value !== "string" || value.trim() === "";
+}
+
+function isValidEmail(value) {
+  return typeof value === "string" && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
+}
