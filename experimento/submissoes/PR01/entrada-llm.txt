+export class BookRepository {
+  #books = [];
+  #nextId = 1;
+
+  async findAll() {
+    return [...this.#books];
+  }
+
+  async findOne(criteria) {
+    const conditions = Object.entries(criteria);
+
+    return (
+      this.#books.find((book) =>
+        conditions.every(([field, value]) => book[field] === value)
+      ) ?? null
+    );
+  }
+
+  async save(book) {
+    if (book.id === undefined) {
+      const createdBook = { ...book, id: this.#nextId++ };
+      this.#books.push(createdBook);
+      return createdBook;
+    }
+
+    const index = this.#books.findIndex((storedBook) => storedBook.id === book.id);
+
+    if (index === -1) {
+      this.#books.push(book);
+    } else {
+      this.#books[index] = book;
+    }
+
+    return book;
+  }
+}

+    const book = await bookService.createBook(req.body ?? {});
+    res.status(201).json(book);
+  });

+  router.get("/", async (req, res) => {
+    const books = await bookService.listBooks();
+    res.json(books);
+  });

+  return router;

+import { AppError } from "../shared/app-error.js";
+
+export class BookService {
+  constructor(bookRepository) {
+    this.bookRepository = bookRepository;
+  }
+
+  async createBook(input) {
+    const { isbn, title, author, totalCopies } = input;
+
+    const existingBook = await this.bookRepository.findOne({ isbn, title });
+
+    if (existingBook) {
+      throw new AppError("ISBN_ALREADY_EXISTS", 409);
+    }
+
+    const book = await this.bookRepository.save({
+      isbn,
+      title,
+      author,
+      totalCopies,
+      availableCopies: input.availableCopies ?? totalCopies,
+    });
+
+    if (isBlank(isbn) || isBlank(title) || isBlank(author)) {
+      throw new AppError("INVALID_BOOK");
+    }
+
+    if (!totalCopies) {
+      throw new AppError("INVALID_QUANTITY");
+    }
+
+    return book;
+  }
+
+  async listBooks() {
+    return this.bookRepository.findAll();
+  }
+}
+
+function isBlank(value) {
+  return typeof value !== "string" || value.trim() === "";
